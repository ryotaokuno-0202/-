    <div class="form-group">
      <label>講義名（任意）</label>
      <input type="text" id="lectureName" placeholder="例：経営学概論 第5回" />
    </div>

    <div class="form-group">
      <label>音声ファイル</label>
      <div class="file-input-wrapper">
        <input type="file" id="audioFile" accept=".wav,.mp3,.m4a,.flac" />
      </div>
      <div class="hint-text">※ 推奨: 20MB以下のMP3/WAV/M4Aファイル。長い講義は分割してください。</div>
    </div>

    <button id="runBtn" class="submit-btn">✨ AIノートを作成する</button>
  </div>

  <div class="output-section">
    <div id="errorArea" style="color: red; font-weight: bold;"></div>
    <h3>文字起こし</h3>
    <div id="transcription" class="output-box"></div>
    <h3>要約</h3>
    <div id="summary" class="output-box"></div>
    <h3>要点</h3>
    <ul id="keyPoints"></ul>
  </div>

  <script>
    // サーバーと通信するためのJavascript（前回のまま中身を維持しています）
    const runBtn = document.getElementById('runBtn');
    const audioFileInput = document.getElementById('audioFile');
    const transcriptionDiv = document.getElementById('transcription');
    const summaryDiv = document.getElementById('summary');
    const keyPointsUl = document.getElementById('keyPoints');
    const errorArea = document.getElementById('errorArea');

    runBtn.addEventListener('click', async () => {
      errorArea.textContent = '';
      const lectureName = document.getElementById('lectureName').value;
      const file = audioFileInput.files[0];

      if (!file) {
        errorArea.textContent = '音声ファイルを選択してください。';
        return;
      }

      const formData = new FormData();
      formData.append('lectureName', lectureName);
      formData.append('audioFile', file);

      try {
        transcriptionDiv.textContent = '処理中...しばらくお待ちください。';
        const res = await fetch('/process', { method: 'POST', body: formData });
        if (!res.ok) {
          errorArea.textContent = 'エラーが発生しました。';
          return;
        }
        const payload = await res.json();
        transcriptionDiv.textContent = payload.transcription || '';
        summaryDiv.textContent = payload.summary || '';
        keyPointsUl.innerHTML = '';
        (payload.keyPoints || []).forEach((kp) => {
          const li = document.createElement('li');
          li.textContent = kp;
          keyPointsUl.appendChild(li);
        });
      } catch (err) {
        errorArea.textContent = 'API呼び出し中にエラーが発生しました。';
      }
    });
  </script>
